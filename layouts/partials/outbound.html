{{- /* The single place that decides where a note sends the reader. The writing is hosted here; films live on YouTube, posts on X and the ventures on their own sites. Every card, list item and single page asks this partial for its URL instead of hard-coding a rule. Priority: YouTube → X → project site → explicit external_url → this site. A Substack external_url no longer sends the reader away: if the note has its text, the essay renders here and the Substack URL comes back as "substack" so the page can offer "read on Substack / subscribe". A Substack note with no body yet still forwards to Substack, which is the safe failure. Returns dict "url", "external" (bool), "platform" (substack|youtube|x|web|site), "label", "substack" (the post's Substack URL, or ""). */ -}} {{- $type := .Params.timeline_type | default "" -}} {{- $url := "" -}} {{- $platform := "site" -}} {{- $substack := "" -}} {{- if .Params.youtube_id -}} {{- $url = printf "https://www.youtube.com/watch?v=%s" .Params.youtube_id -}} {{- $platform = "youtube" -}} {{- else if and (eq .Params.platform "x") .Params.author_handle .Params.post_id -}} {{- $url = printf "https://x.com/%s/status/%s" .Params.author_handle .Params.post_id -}} {{- $platform = "x" -}} {{- else if and (eq $type "project") .Params.website_url -}} {{- $url = .Params.website_url -}} {{- $platform = "web" -}} {{- else -}} {{- with .Params.external_url -}} {{- $url = . -}} {{- if findRE `substack\.com` . 1 -}} {{- $substack = . -}} {{- if strings.TrimSpace $.RawContent -}} {{- $platform = "site" -}} {{- else -}} {{- $platform = "substack" -}} {{- end -}} {{- else if findRE `(youtube\.com|youtu\.be)` . 1 -}} {{- $platform = "youtube" -}} {{- else if findRE `(^|//|\.)x\.com` . 1 -}} {{- $platform = "x" -}} {{- else -}} {{- $platform = "web" -}} {{- end -}} {{- end -}} {{- end -}} {{- $labels := dict "substack" "Substack" "youtube" "YouTube" "x" "X" "web" "its own site" "site" "this site" -}} {{- $external := ne $platform "site" -}} {{- if not $external -}} {{- $url = .RelPermalink -}} {{- end -}} {{- return (dict "url" $url "external" $external "platform" $platform "label" (index $labels $platform) "substack" $substack) -}}
